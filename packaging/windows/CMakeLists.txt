cmake_minimum_required(VERSION 3.24)
project(DocumentViewerPortable LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/EnableLto.cmake")
if(NOT EXISTS "${PAYLOAD}")
    message(FATAL_ERROR "Set PAYLOAD to the deployed application ZIP.")
endif()
file(TO_CMAKE_PATH "${PAYLOAD}" PAYLOAD)
configure_file(payload.rc.in payload.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/payload.rc" PROPERTIES OBJECT_DEPENDS "${PAYLOAD}")
set(APP_ICON "${CMAKE_CURRENT_LIST_DIR}/../../resources/icons/DocumentViewer.ico")
configure_file("${CMAKE_CURRENT_LIST_DIR}/../../resources/app-icon.rc.in" app-icon.rc @ONLY)
set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/app-icon.rc" PROPERTIES OBJECT_DEPENDS "${APP_ICON}")
add_executable(DocumentViewerPortable WIN32 launcher.cpp
    "${CMAKE_CURRENT_BINARY_DIR}/payload.rc" "${CMAKE_CURRENT_BINARY_DIR}/app-icon.rc")
target_link_libraries(DocumentViewerPortable PRIVATE shell32 ole32)
if(MINGW)
    target_link_options(DocumentViewerPortable PRIVATE -static -static-libgcc -static-libstdc++)
endif()
